    '''expr : expr OR term'''
    p[0] = p[1] or p[3]

def p_expr_term(p):
    '''expr : term'''
    p[0] = p[1]

def p_term_and(p):
    '''term : term AND factor'''
    p[0] = p[1] and p[3]

def p_term_factor(p):
    '''term : factor'''
    p[0] = p[1]

def p_factor(p):
    '''factor : NOT factor
              | TRUE
              | FALSE
              | LPAREN expr RPAREN'''
    if len(p) == 3:          # NOT factor
        p[0] = not p[2]
    elif p[1] == 'TRUE':
        p[0] = True
    elif p[1] == 'FALSE':
        p[0] = False
    else:                       # ( expr )
        p[0] = p[2]

    action    = p[4]
    if condition and action is not None:
        print(action)

    '''cond : NUMBER GT NUMBER
           | NUMBER LT NUMBER
           | NUMBER EQ NUMBER'''
    if   p[2] == '>':  p[0] = p[1] >  p[3]

    '''expr : NUMBER op NUMBER'''
    ops = {
        '==': lambda a, b: a == b,

        '<=': lambda a, b: a <= b,

## Boolean Expression Lexer and Parser
Problem Statement

Build a Lexer and Parser for evaluating simple Boolean expressions.

The parser must correctly handle operator precedence in the following order:

Required Tokens,TRUE,FALSE,AND,OR,NOT,LPAREN,RPAREN

All keywords are reserved words extracted from identifiers.


NOT → Highest precedence
AND → Medium precedence
OR → Lowest precedence

Parentheses () must override precedence normally.
expr   : expr OR term
       | term

term   : term AND factor
       | factor

factor : NOT factor
       | TRUE
       | FALSE
       | LPAREN expr RPAREN

TRUE AND FALSE OR TRUE
NOT FALSE AND TRUE
TRUE OR (FALSE AND TRUE)
NOT (TRUE OR FALSE)

Expected Output
Result: True
Result: True
Result: True
Result: False

اسم الملف: IF–THEN Conditional Language
Build a Lexer and Parser for a simple conditional language using the syntax:
IF <condition> THEN <action>. The condition compares two numbers (e.g., x > 5),
and the action is either PRINT "message" or SKIP. The parser should evaluate each statement and execute the action only if the condition is true.

Required Tokens
IF
THEN
PRINT
SKIP
NUMBER
STRING
GT
LT
EQ
All keywords are reserved. EQ matches ==.

Grammar Rules
program : program stmt | stmt stmt : IF cond THEN action cond : NUMBER GT NUMBER | NUMBER LT NUMBER | NUMBER EQ NUMBER action : PRINT STRING | SKIP

Sample Input
IF 10 > 3 THEN PRINT "Big" IF 5 > 9 THEN PRINT "Unreachable" IF 7 == 7 THEN PRINT "Equal" IF 2 < 1 THEN SKIP

Expected Output
Big Equal
Hint: Use empty productions pattern knowledge from Section 4. The action rule for SKIP simply does nothing.
Store the condition result in p[0] and pass it up to stmt.
